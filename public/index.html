copyBtn.onclick = async () => { const link = row.dataset.url || input.value; try{ await navigator.clipboard.writeText(link); }catch{ const temp = document.createElement("textarea"); temp.value = link; temp.style.position = "fixed"; temp.style.opacity = "0"; document.body.appendChild(temp); temp.select(); document.execCommand("copy"); temp.remove(); } copyBtn.classList.add("done"); clearTimeout(copyBtn._t); copyBtn._t = setTimeout(() => copyBtn.classList.remove("done"), 1600); }; openBtn.onclick = () => window.open( row.dataset.url || input.value, "_blank", "noopener,noreferrer" );
